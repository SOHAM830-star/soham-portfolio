import { useEffect, useRef, useCallback } from 'react';

// Zero-copy WebSocket telemetry stream receiver with binary chunking
export function useReactiveStream<T>(endpoint: string, onFrame: (data: T) => void) {
  const socketRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    const ws = new WebSocket(endpoint);
    ws.binaryType = 'arraybuffer';

    ws.onmessage = (event: MessageEvent) => {
      const buffer = new Float32Array(event.data);
      onFrame(buffer as unknown as T);
    };

    socketRef.current = ws;
    return () => ws.close();
  }, [endpoint, onFrame]);
}
use std::sync::atomic::{AtomicUsize, Ordering};

// Lock-free circular ring buffer for microsecond tick matching
pub struct LockFreeRingBuffer<T, const CAP: usize> {
    buffer: [Option<T>; CAP],
    head: AtomicUsize,
    tail: AtomicUsize,
}

impl<T: Copy, const CAP: usize> LockFreeRingBuffer<T, CAP> {
    pub fn push(&mut self, item: T) -> Result<(), &'static str> {
        let current_head = self.head.load(Ordering::Relaxed);
        self.buffer[current_head % CAP] = Some(item);
        self.head.store(current_head + 1, Ordering::Release);
        Ok(())
    }
}
